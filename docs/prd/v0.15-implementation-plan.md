## Task 3: Schema 枚举 CHECK + 唯一性 + 索引 Migration

**Files:**

- Create: `migrations/002_enums_constraints_indexes.ts`
- **Step 1: 创建 migration 文件**

此 migration 包含三部分：

1. CHECK 约束（通过重建表实现，因为 SQLite 不支持 ALTER TABLE ADD CHECK）
2. 唯一性约束（partial index for employee.code）
3. 查询索引

**关键实现注意**：

- 重建 `employees` 表时，先验证无违规 status 值：`SELECT DISTINCT status FROM employees WHERE status NOT IN ('active','inactive','archived')`
- 重建 `run_records` 表时同理
- 使用事务保护整个 migration

```typescript
import type { Knex } from "knex";

export async function up(knex: Knex): Promise<void> {
  // Part 1: Validate existing data before rebuild
  const badEmployeeStatus = await knex.raw(
    `SELECT COUNT(*) as cnt FROM employees WHERE status NOT IN ('active','inactive','archived')`
  );
  if (badEmployeeStatus[0]?.cnt > 0) {
    await knex.raw(`UPDATE employees SET status = 'active' WHERE status NOT IN ('active','inactive','archived')`);
  }

  const badRunStatus = await knex.raw(
    `SELECT COUNT(*) as cnt FROM run_records WHERE status NOT IN ('pending','running','completed','failed')`
  );
  if (badRunStatus[0]?.cnt > 0) {
    await knex.raw(`UPDATE run_records SET status = 'pending' WHERE status NOT IN ('pending','running','completed','failed')`);
  }

  // Part 2: Partial UNIQUE index for employee.code (active only)
  await knex.raw(`CREATE UNIQUE INDEX IF NOT EXISTS idx_employees_code_active ON employees(code) WHERE status != 'archived'`);

  // Part 3: Composite UNIQUE for employee_skills
  await knex.raw(`CREATE UNIQUE INDEX IF NOT EXISTS idx_employee_skills_unique ON employee_skills(employee_id, skill_name)`);

  // Part 4: Query indexes
  await knex.raw(`CREATE INDEX IF NOT EXISTS idx_run_records_employee_started ON run_records(employee_id, started_at DESC)`);
  await knex.raw(`CREATE INDEX IF NOT EXISTS idx_run_events_run_seq ON run_events(run_id, seq ASC)`);
  await knex.raw(`CREATE INDEX IF NOT EXISTS idx_schedule_jobs_employee ON employee_schedule_jobs(employee_id)`);
}

export async function down(knex: Knex): Promise<void> {
  await knex.raw(`DROP INDEX IF EXISTS idx_employees_code_active`);
  await knex.raw(`DROP INDEX IF EXISTS idx_employee_skills_unique`);
  await knex.raw(`DROP INDEX IF EXISTS idx_run_records_employee_started`);
  await knex.raw(`DROP INDEX IF EXISTS idx_run_events_run_seq`);
  await knex.raw(`DROP INDEX IF EXISTS idx_schedule_jobs_employee`);
}
```

> **注意**：暂不做 CHECK 约束的表重建（风险高、收益低），改为在 TypeScript 层保证枚举安全。CHECK 约束推迟到数据充分验证后。

```typescript
export type PlatformErrorCode =
  | "config_error"
  | "integration_error"
  | "runtime_error"
  | "model_error"
  | "data_error";

export class PlatformError extends Error {
  constructor(
    public readonly code: PlatformErrorCode,
    message: string,
    public readonly userHint: string,
    public readonly recoveryAction?: string
  ) {
    super(message);
    this.name = "PlatformError";
  }

  toJSON() {
    return {
      code: this.code,
      message: this.message,
      userHint: this.userHint,
      recoveryAction: this.recoveryAction,
    };
  }
}

export class ConfigError extends PlatformError {
  constructor(message: string, userHint: string, recoveryAction?: string) {
    super("config_error", message, userHint, recoveryAction);
    this.name = "ConfigError";
  }
}

export class IntegrationError extends PlatformError {
  constructor(message: string, userHint: string, recoveryAction?: string) {
    super("integration_error", message, userHint, recoveryAction);
    this.name = "IntegrationError";
  }
}

export class RuntimeError extends PlatformError {
  constructor(message: string, userHint: string, recoveryAction?: string) {
    super("runtime_error", message, userHint, recoveryAction);
    this.name = "RuntimeError";
  }
}

export class ModelError extends PlatformError {
  constructor(message: string, userHint: string, recoveryAction?: string) {
    super("model_error", message, userHint, recoveryAction);
    this.name = "ModelError";
  }
}

export class DataError extends PlatformError {
  constructor(message: string, userHint: string, recoveryAction?: string) {
    super("data_error", message, userHint, recoveryAction);
    this.name = "DataError";
  }
}

export function classifyError(error: unknown): PlatformError {
  if (error instanceof PlatformError) return error;

  const msg = error instanceof Error ? error.message : String(error);
  const lower = msg.toLowerCase();

  if (lower.includes("api key") || lower.includes("apikey") || lower.includes("not configured")) {
    return new ConfigError(msg, "请先在设置中配置模型 API Key", "navigate:/integrations");
  }
  if (lower.includes("timeout") || lower.includes("timed out")) {
    return new RuntimeError(msg, "执行超时，请稍后重试");
  }
  if (lower.includes("connection") || lower.includes("econnrefused") || lower.includes("fetch failed")) {
    return new IntegrationError(msg, "连接异常，请检查网络或集成配置", "navigate:/integrations");
  }
  if (lower.includes("model") || lower.includes("invalid response") || lower.includes("parse")) {
    return new ModelError(msg, "模型响应异常，请检查模型配置", "navigate:/integrations");
  }

  return new RuntimeError(msg, "执行异常，请查看运行详情排查问题");
}
```

      const wsFiles = input.workspaceFiles ?? {};
      const wsDir = resolveEmployeeWorkspace(this.gateway.homeDir, employee.code);
      for (const [filename, content] of Object.entries(wsFiles)) {
        if (WRITABLE_FILES.has(filename) && typeof content === "string" && content.trim()) {
          writeFileSync(join(wsDir, filename), content, "utf-8");
        }
      }

      // Step 4: Skill bind
      const skills = await this.employeeSkillRepo.replaceForEmployee(
        employee.id,
        input.skillNames ?? []
      );
      completedSteps.push({
        name: "skill-bind",
        rollback: async () => { await this.employeeSkillRepo.replaceForEmployee(employee.id, []); }
      });

        (input.scheduleKind === "cron" ? input.cronExpr : input.everyMs);

    } catch (error) {

      for (const step of completedSteps.reverse()) {
        try { await step.rollback(); } catch { /* best-effort */ }
      }
      throw error;
    }

  async deleteEmployee(id: string): Promise<{ id: string; code: string }> {

    if (!employee) throw new Error(`Employee not found: ${id}`);

const CURRENT_KEY_VERSION = 0x01;

function encrypt(plaintext: string, key: Buffer): string {
  const iv = randomBytes(IV_LENGTH);
  const cipher = createCipheriv(ALGORITHM, key, iv);
  const encrypted = Buffer.concat([cipher.update(plaintext, "utf-8"), cipher.final()]);
  const authTag = cipher.getAuthTag();
  const version = Buffer.alloc(1, CURRENT_KEY_VERSION);
  return Buffer.concat([version, iv, authTag, encrypted]).toString("base64");
}

function decrypt(ciphertext: string, key: Buffer): string {
  const data = Buffer.from(ciphertext, "base64");

  // Detect version: new format starts with version byte (0x01).
  // Old format length: iv(16) + authTag(16) + encrypted(N) where N > 0
  // Minimum old format: 33 bytes. New format: 34+ bytes.
  // We distinguish by checking: if first byte is a known version AND length is odd
  // (old format always iv+tag+cipher, new has extra byte making total parity differ).
  // Safest approach: try versioned first, fall back to old format on decrypt failure.
  let offset = 0;
  const isVersioned = data.length > 1 + IV_LENGTH + AUTH_TAG_LENGTH && data[0] === CURRENT_KEY_VERSION;
  if (isVersioned) {
    try {
      return decryptWithOffset(data, key, 1);
    } catch {
      // Fall through to old format
    }
  }
  return decryptWithOffset(data, key, 0);
}

function decryptWithOffset(data: Buffer, key: Buffer, offset: number): string {

  const iv = data.subarray(offset, offset + IV_LENGTH);
  const authTag = data.subarray(offset + IV_LENGTH, offset + IV_LENGTH + AUTH_TAG_LENGTH);
  const encrypted = data.subarray(offset + IV_LENGTH + AUTH_TAG_LENGTH);
  const decipher = createDecipheriv(ALGORITHM, key, iv);
  decipher.setAuthTag(authTag);
  return Buffer.concat([decipher.update(encrypted), decipher.final()]).toString("utf-8");
}

```typescript
// server/services/employee-health-service.ts
import type { HealthStatus, HealthDetail } from "../../shared/ui-models";

type EmployeeHealthInput = {
  employeeId: string;
  status: string;
  skillCount: number;
  recentRuns: Array<{ status: string; startedAt: string }>;
  integrationIssues: string[];
};

export class EmployeeHealthService {
  private readonly CONSECUTIVE_FAILURE_THRESHOLD = 3;
  private readonly IDLE_THRESHOLD_MS = 7 * 24 * 60 * 60 * 1000; // 7 days

  computeHealth(input: EmployeeHealthInput): { healthStatus: HealthStatus; healthDetail: HealthDetail } {
    const lastRun = input.recentRuns[0];
    const lastRunAt = lastRun?.startedAt ?? null;

    const consecutiveFailures = this.countConsecutiveFailures(input.recentRuns);

    const detail: HealthDetail = {
      reason: "ok",
      lastRunAt,
      consecutiveFailures,
      missingIntegrations: input.integrationIssues,
    };

    if (input.skillCount === 0) {
      return { healthStatus: "unconfigured", healthDetail: { ...detail, reason: "no_skills" } };
    }

    if (input.status === "inactive") {
      return { healthStatus: "inactive", healthDetail: { ...detail, reason: "disabled" } };
    }

    if (consecutiveFailures >= this.CONSECUTIVE_FAILURE_THRESHOLD) {
      return { healthStatus: "error", healthDetail: { ...detail, reason: "consecutive_failures" } };
    }

    if (input.integrationIssues.length > 0) {
      return { healthStatus: "warning", healthDetail: { ...detail, reason: "integration_failed" } };
    }

    if (lastRunAt && Date.now() - new Date(lastRunAt).getTime() > this.IDLE_THRESHOLD_MS) {
      return { healthStatus: "warning", healthDetail: { ...detail, reason: "idle_too_long" } };
    }

    if (lastRun?.status === "completed") {
      return { healthStatus: "healthy", healthDetail: { ...detail, reason: "recent_success" } };
    }

    return { healthStatus: "healthy", healthDetail: detail };
  }

  computeHealthBatch(inputs: EmployeeHealthInput[]): Map<string, { healthStatus: HealthStatus; healthDetail: HealthDetail }> {
    const results = new Map();
    for (const input of inputs) {
      results.set(input.employeeId, this.computeHealth(input));
    }
    return results;
  }

  private countConsecutiveFailures(runs: Array<{ status: string }>): number {
    let count = 0;
    for (const run of runs) {
      if (run.status === "failed") count++;
      else break;
    }
    return count;
  }
}

    const healthResult = healthMap.get(employee.id);

    return {
      ...employee,
      skills,
      schedule,
      latestRun: runs[0] ?? null,
      jobsCount: jobs.length,
      enabledJobsCount: jobs.filter(j => j.enabled).length,
      // Legacy health (backward compat)
      health: {
        hasSkills: skills.length > 0,
        hasSchedule: Boolean(schedule) || jobs.length > 0,
        lastStatus: runs[0]?.status ?? "idle",
      },
      // New health model
      healthStatus: healthResult?.healthStatus ?? "unconfigured",
      healthDetail: healthResult?.healthDetail ?? null,
    };
  });

  return { ok: true, data: enriched };
});
```

- **Step 6: 注册 EmployeeHealthService 到 PlatformContext**

在 `platform-context.ts` 中创建并注册 `EmployeeHealthService` 实例。
